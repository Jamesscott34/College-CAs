    Name VARCHAR(35),
    
    -- Define a column for movie genres
    Genre VARCHAR(20),

    Duration VARCHAR(10),
    
    -- Define a column for movie qualifications (ratings)
    Qualification INT,
    
    -- Define a column for protagonist actor names

-- Create a table named MoviesActivity to track user activity related to movies
CREATE TABLE MoviesActivity (
    -- Define a column for usernames
    Username VARCHAR(35),
    
    -- Define a column for movie IDs
    Movie_ID INT,
    
    -- Define a column for movie genres
    Genre VARCHAR(20),
    
    -- Define a composite primary key using Username and Movie_ID
    PRIMARY KEY (Username, Movie_ID),
    
    -- Define a foreign key constraint referencing the UserName column in the Users table
    FOREIGN KEY (Username) REFERENCES Users(UserName),
    
    -- Define a foreign key constraint referencing the ID_movie column in the Movies table
    FOREIGN KEY (Movie_ID) REFERENCES Movies(ID_movie)
);


-- Insert data into MoviesActivity with random Movie_ID and corresponding genre for each user
INSERT INTO MoviesActivity (Username, Movie_ID, Genre)
SELECT 
    u.Username,
    (
        SELECT ID_movie
        FROM Movies
        ORDER BY RAND()
        LIMIT 1
    ) AS Movie_ID,
    (
        SELECT Genre
        FROM Movies
        ORDER BY RAND()
        LIMIT 1
    ) AS Genre
FROM
    Users u;

-- Create a table named SeriesActivity to track user activity related to TV series
CREATE TABLE SeriesActivity (
    -- Define a column for usernames
    Username VARCHAR(35),
    
    -- Define a column for series IDs
    Series_ID INT,
    
    -- Define a column for series genres
    Genre VARCHAR(20),
    
    -- Define a composite primary key using Username and Series_ID
    PRIMARY KEY (Username, Series_ID),
    
    -- Define a foreign key constraint referencing the UserName column in the Users table
    FOREIGN KEY (Username) REFERENCES Users(UserName),
    
    -- Define a foreign key constraint referencing the ID_series column in the Series table
    FOREIGN KEY (Series_ID) REFERENCES Series(ID_series)
);


-- Insert data into SeriesActivities with random Series_ID and corresponding genre for each user
INSERT INTO SeriesActivity (Username, Series_ID, Genre)
SELECT 
    u.Username,
    (
        SELECT ID_series
        FROM Series
        ORDER BY RAND()
        LIMIT 1
    ) AS Series_ID,
    (
        SELECT Genre
        FROM Series
        ORDER BY RAND()
        LIMIT 1
    ) AS Genre
FROM
    Users u;
